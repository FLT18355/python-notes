# 类与面向对象

## 基本概念

面向对象编程（OOP）把数据和操作封装在"对象"中，核心概念：
- **类（Class）**：对象的模板
- **对象（Object）**：类的实例
- **属性（Attribute）**：对象的数据
- **方法（Method）**：对象的行为

## 定义类

```python
class Dog:
    # 构造方法，创建对象时自动调用
    def __init__(self, name, age):
        self.name = name  # 实例属性
        self.age = age

    # 实例方法
    def bark(self):
        return f"{self.name}：汪汪！"

    def get_info(self):
        return f"{self.name}，{self.age} 岁"


# 创建对象（实例化）
dog = Dog("旺财", 3)
print(dog.bark())        # 旺财：汪汪！
print(dog.get_info())    # 旺财，3 岁
print(dog.name)          # 旺财（访问属性）
```

## self 关键字

`self` 代表实例本身，调用方法时自动传入：

```python
class Counter:
    def __init__(self):
        self.count = 0

    def increment(self):
        self.count += 1
        return self.count


c = Counter()
print(c.increment())  # 1
print(c.increment())  # 2
```

## 类属性与实例属性

```python
class Dog:
    species = "犬科"  # 类属性（所有实例共享）

    def __init__(self, name):
        self.name = name  # 实例属性（每个实例独立）


dog1 = Dog("旺财")
dog2 = Dog("小白")

print(dog1.species)  # 犬科
print(dog2.species)  # 犬科
print(dog1.name)     # 旺财
print(dog2.name)     # 小白

# 修改类属性影响所有实例
Dog.species = "犬科动物"
print(dog1.species)  # 犬科动物
```

## 继承

```python
class Animal:
    def __init__(self, name):
        self.name = name

    def speak(self):
        return "声音"


class Dog(Animal):
    def speak(self):  # 重写（override）
        return f"{self.name}：汪汪！"


class Cat(Animal):
    def speak(self):
        return f"{self.name}：喵喵~"


dog = Dog("旺财")
cat = Cat("咪咪")

print(dog.speak())  # 旺财：汪汪！
print(cat.speak())  # 咪咪：喵喵~
```

### super() 调用父类

```python
class Animal:
    def __init__(self, name, age):
        self.name = name
        self.age = age


class Dog(Animal):
    def __init__(self, name, age, breed):
        super().__init__(name, age)  # 调用父类构造方法
        self.breed = breed

    def info(self):
        return f"{self.name}，{self.age} 岁，{self.breed}"


dog = Dog("旺财", 3, "金毛")
print(dog.info())  # 旺财，3 岁，金毛
```

## 多态

不同类的对象对同一方法做出不同响应：

```python
def make_sound(animal):
    print(animal.speak())


make_sound(Dog("旺财"))   # 旺财：汪汪！
make_sound(Cat("咪咪"))   # 咪咪：喵喵~
```

## 封装

使用私有属性保护数据：

```python
class BankAccount:
    def __init__(self, owner, balance):
        self.owner = owner       # 公开
        self._balance = balance  # 受保护（约定俗成）
        self.__password = "1234"  # 私有（名称改写）

    def deposit(self, amount):
        if amount > 0:
            self._balance += amount
        return self._balance

    def get_balance(self):
        return self._balance


account = BankAccount("张三", 1000)
print(account.get_balance())    # 1000
print(account._balance)         # 1000（可以访问，但不推荐）
# print(account.__password)     # ❌ AttributeError
```

## 类方法与静态方法

```python
class MyClass:
    class_attr = "类属性"

    # 实例方法：第一个参数是 self
    def instance_method(self):
        return f"实例方法，{self.class_attr}"

    # 类方法：第一个参数是 cls，可以访问类属性
    @classmethod
    def class_method(cls):
        return f"类方法，{cls.class_attr}"

    # 静态方法：不需要 self 或 cls
    @staticmethod
    def static_method():
        return "静态方法"


obj = MyClass()
print(obj.instance_method())  # 实例方法，类属性
print(MyClass.class_method())  # 类方法，类属性
print(MyClass.static_method())  # 静态方法
```

## 属性（property）

```python
class Circle:
    def __init__(self, radius):
        self._radius = radius

    @property
    def radius(self):
        return self._radius

    @radius.setter
    def radius(self, value):
        if value < 0:
            raise ValueError("半径不能为负")
        self._radius = value

    @property
    def area(self):
        import math
        return math.pi * self._radius ** 2


circle = Circle(5)
print(circle.radius)   # 5（像属性一样访问）
print(circle.area)     # 78.54

circle.radius = 10     # 调用 setter
print(circle.area)     # 314.16
```

## 特殊方法（魔术方法）

```python
class Vector:
    def __init__(self, x, y):
        self.x = x
        self.y = y

    def __str__(self):
        return f"Vector({self.x}, {self.y})"

    def __repr__(self):
        return f"Vector(x={self.x}, y={self.y})"

    def __add__(self, other):
        return Vector(self.x + other.x, self.y + other.y)

    def __len__(self):
        return 2

    def __eq__(self, other):
        return self.x == other.x and self.y == other.y


v1 = Vector(1, 2)
v2 = Vector(3, 4)
v3 = v1 + v2

print(str(v1))   # Vector(1, 2)
print(repr(v1))  # Vector(x=1, y=2)
print(len(v1))   # 2
print(v1 == v2)  # False
```

## 常用魔术方法

| 方法 | 调用时机 |
|------|----------|
| `__init__` | 创建对象时 |
| `__str__` | str() / print() |
| `__repr__` | repr() / 交互式解释器 |
| `__len__` | len() |
| `__eq__` | == |
| `__lt__` | < |
| `__add__` | + |
| `__getitem__` | obj[key] |
| `__setitem__` | obj[key] = value |

## 抽象基类

```python
from abc import ABC, abstractmethod


class Shape(ABC):
    @abstractmethod
    def area(self):
        pass


class Circle(Shape):
    def __init__(self, radius):
        self.radius = radius

    def area(self):
        import math
        return math.pi * self.radius ** 2


c = Circle(5)
print(c.area())  # 78.54
```
