# 常用内置函数

## 输入输出

```python
print("hello")           # 输出
input("请输入：")        # 输入（返回字符串）
```

## 类型相关

```python
type(10)                 # <class 'int'>
isinstance(10, int)      # True
issubclass(bool, int)    # True
```

## 类型转换

```python
int("123")               # 123
float("3.14")            # 3.14
str(100)                 # "100"
bool(0)                  # False
list("hello")            # ['h', 'e', 'l', 'l', 'o']
tuple([1, 2, 3])         # (1, 2, 3)
set([1, 2, 2, 3])        # {1, 2, 3}
dict([("a", 1), ("b", 2)])  # {'a': 1, 'b': 2}
```

## 数值函数

```python
abs(-5)                  # 5（绝对值）
round(3.14159, 2)        # 3.14（四舍五入）
pow(2, 3)                # 8（幂运算）
divmod(10, 3)            # (3, 1)（商和余）
min(3, 1, 4)             # 1
max(3, 1, 4)             # 4
sum([1, 2, 3, 4])        # 10
```

## 序列函数

```python
len([1, 2, 3])           # 3（长度）
sorted([3, 1, 4, 2])     # [1, 2, 3, 4]（返回新列表）
reversed([1, 2, 3])      # <list_reverseiterator>（反转）
list(reversed([1, 2, 3])) # [3, 2, 1]

# all 和 any
all([True, True, True])  # True（全部为真）
any([False, True, False]) # True（任一为真）
```

## 枚举与拉链

```python
# enumerate — 带索引遍历
fruits = ["苹果", "香蕉", "橙子"]
for i, fruit in enumerate(fruits):
    print(f"{i}: {fruit}")

# zip — 同时遍历多个序列
names = ["张三", "李四", "王五"]
ages = [25, 30, 35]
for name, age in zip(names, ages):
    print(f"{name}: {age}岁")

# zip 解包
pairs = [(1, "a"), (2, "b"), (3, "c")]
nums, letters = zip(*pairs)
print(nums)     # (1, 2, 3)
print(letters)  # ('a', 'b', 'c')
```

## map、filter、reduce

```python
# map — 对每个元素应用函数
nums = [1, 2, 3, 4]
squares = list(map(lambda x: x ** 2, nums))
print(squares)  # [1, 4, 9, 16]

# filter — 过滤
evens = list(filter(lambda x: x % 2 == 0, nums))
print(evens)  # [2, 4]

# reduce — 累积（需要从 functools 导入）
from functools import reduce
result = reduce(lambda a, b: a + b, nums)
print(result)  # 10（1+2+3+4）
```

## 其他

```python
# id — 对象内存地址
a = [1, 2, 3]
b = a
print(id(a) == id(b))  # True

# hash — 哈希值
print(hash("hello"))   # 哈希值
print(hash(123))       # 123

# help — 帮助
help(str.upper)

# dir — 查看属性和方法
print(dir([1, 2, 3]))

# exec — 执行字符串代码
exec("x = 10\nprint(x)")  # 10

# eval — 计算表达式
print(eval("1 + 2 * 3"))  # 7

# ord / chr — 字符编码
print(ord("A"))   # 65
print(chr(65))    # 'A'

# bin / oct / hex — 进制转换
print(bin(10))    # '0b1010'
print(oct(10))    # '0o12'
print(hex(10))    # '0xa'
```

## 实战示例

```python
# 统计成绩
scores = [85, 92, 78, 90, 65]
print(f"最高分：{max(scores)}")
print(f"最低分：{min(scores)}")
print(f"平均分：{sum(scores) / len(scores):.1f}")
print(f"及格率：{sum(1 for s in scores if s >= 60) / len(scores) * 100:.1f}%")

# 数据处理
users = [
    {"name": "张三", "age": 25, "city": "北京"},
    {"name": "李四", "age": 30, "city": "上海"},
    {"name": "王五", "age": 22, "city": "北京"},
]

# 筛选北京用户
beijing = list(filter(lambda u: u["city"] == "北京", users))
print(f"北京用户：{[u['name'] for u in beijing]}")

# 按年龄排序
sorted_users = sorted(users, key=lambda u: u["age"])
print(f"按年龄排序：{[u['name'] for u in sorted_users]}")
```
